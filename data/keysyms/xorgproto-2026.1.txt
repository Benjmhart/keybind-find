# commit 6d3907b6d5dc4aa7d7ee6ee5d7d8714e7466630b), in three sections, each generated in include/X11:
#
# 1. `name 0xvalue` from keysymdef.h:
#   sed -n 's/^#define XK_\([A-Za-z0-9_]*\) *\(0x[0-9a-fA-F]*\).*/\1 \2/p' keysymdef.h
# 2. `XF86name 0xvalue` from XF86keysym.h (XF86XK_Foo written XF86Foo, as xkbcommon names it; _EVDEVK(v) is
#    0x10081000 + v):
#   awk '/^#define XF86XK_/ { v = $3; if (v ~ /^_EVDEVK\(/) { gsub(/^_EVDEVK\(|\)$/, "", v); v = sprintf("0x%x", 0x10081000 + strtonum(v)) } print "XF86" substr($2, 8), v }' XF86keysym.h
# 3. `U+XXXX 0xvalue`: the named keysym for each code point outside U+0020-U+007E and U+00A0-U+00FF, from
#    keysymdef.h's one-to-one `/* U+XXXX NAME */` comments (not the deprecated `/*(U+...)*/` or the more specific
#    `/*<U+...>*/`), the lowest keysym when several share a code point and none of the 0x01000000 + code point form, as
#    xkbcommon's xkb_utf32_to_keysym chooses (its keysym-utf.c table at xkbcommon-1.13.2 gives the same 722 named
#    keysyms, and its special cases the 7 controls here):
#   awk 'match($0, /^#define XK_[A-Za-z0-9_]+ +(0x[0-9a-fA-F]+) +\/\* U\+([0-9A-F]+) /, m) { cp = strtonum("0x" m[2]); k = strtonum(m[1]); if ((cp < 0x20 || cp > 0x7e) && (cp < 0xa0 || cp > 0xff) && k < 0x1000000 && (!(cp in best) || k < best[cp])) best[cp] = k } END { for (cp in best) printf "U+%04X 0x%x\n", cp, best[cp] }' keysymdef.h | LC_ALL=C sort
